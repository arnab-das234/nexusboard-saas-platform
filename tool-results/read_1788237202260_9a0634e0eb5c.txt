    94→// Simulated page content lines for the PDF placeholder
    95→function generatePageLines(page: number) {
    96→  const title = page === 1;
    97→  const lines: { w: number; h: number; mt: boolean }[] = [];
    98→  const seed = page * 7;
    99→  if (title) {
   100→    lines.push({ w: 65, h: 12, mt: true });
   101→    lines.push({ w: 40, h: 8, mt: true });
   102→    lines.push({ w: 0, h: 16, mt: false });
   103→  }
   104→  const lineCount = title ? 18 : 22;
   105→  for (let i = 0; i < lineCount; i++) {
   106→    const variance = ((seed + i * 13) % 30) - 10;
   107→    lines.push({ w: Math.max(50, Math.min(95, 80 + variance)), h: 5, mt: i === 0 && !title });
   108→  }
   109→  return lines;
   110→}

   163→      const range = sel.getRangeAt(0);
   164→      const rect = range.getBoundingClientRect();
   165→      const viewerRect = viewerRef.current?.getBoundingClientRect();
   166→      if (viewerRect) {
   167→        setCopyPos({
   168→          x: rect.left - viewerRect.left + rect.width / 2,
   169→          y: rect.top - viewerRect.top - 10,
   170→        });
   171→      }
   172→      setSelectedText(sel.toString().trim());
   173→      setShowCopy(true);
   174→    } else {
   175→      setShowCopy(false);
   176→    }
   177→  }
   178→

   180→ navigator.clipboard.writeText(selectedText);
   181→    onTextSelect(selectedText);
   182→    setShowCopy(false);
   183→    toast.success('Selection copied and added to references');
   184→    window.getSelection()?.removeAllRanges();
   185→  }
   186→
   187→  if (mobileTab && mobileTab !== 'viewer') return null;
   188→
   189→  return (
   190→    <div className="flex flex-col h-full">
   191→      {/* Viewer Toolbar */}
   192→      <div className="flex items-center justify-between gap-2 border-b border-slate-200 bg-slate-50 px-3 py-2 shrink-0">

   203→          <TooltipProvider><Tooltip><TooltipTrigger asChild>
   204→            <Button variant="ghost" size="icon" className="h-7 w-7" onClick={() => setZoom(Math.max(50, zoom - 25))}>
   205→              <ZoomOut className="h-3.5 w-3.5" />
   206→            </Button>
   207→          </TooltipTrigger><TooltipContent>Zoom Out</TooltipContent></Tooltip></TooltipProvider>
   208→          <span className="text-xs font-mono text-slate-500 w-10 text-center">{zoom}%</span>
   209→          <TooltipProvider><Tooltip><TooltipTrigger asChild>
   210→            <Button variant="ghost" size="icon" className="h-7 w-7" onClick={() => setZoom(Math.min(200, zoom + 25))}>
   211→              <ZoomIn className="h-3.5 w-3.5" />
   212→            </Button>
   213→          </TooltipTrigger><TooltipContent>Zoom In</TooltipContent></Tooltip></TooltipProvider>
   214→          <Separator orientation="vertical" className="h-4 mx-1" />
   215→          <Button variant="ghost" size="icon" className="h-7 w-7" onClick={() => setCurrentPage(Math.max(1, currentPage - 1))} disabled={currentPage <= 1}>
   216→            <ChevronLeft className="h-3.5 w-3.5" />
   217→          </Button>
   218→          <span className="text-xs font-mono text-slate-500">{currentPage} / {essay.totalPages}</span>
   219→          <Button variant="ghost" size="icon" className="h-7 w-7" onClick={() => setCurrentPage(Math.min(essay.totalPages, currentPage + 1))} disabled={currentPage >= essay.totalPages}>
   220→            <ChevronRight className="h-3.5 w-3.5" />

   222→          <Separator orientation="vertical" className="h-4 mx-1" />

   232→          </TooltipTrigger><TooltipContent>Download</TooltipContent></Tooltip></TooltipProvider>

   236→      {/* PDF Content Area */}

   295→// ── Evaluation Form ──────────────────────────────────────────────────────────
   296→function EvaluationForm({
   297→  essay, scores, setScores, justification, setJustification,
   298→  comments, setComments, quotes, setQuotes, isSubmitted,
   299→  onSaveDraft, onSubmit, saving, mobileTab
   300→}: {
   301→  essay: EssayDetail;
   302→  scores: Record<string, number>;
   303→  setScores: React.Dispatch<React.SetStateAction<Record<string, number>>>;
   304→  justification: string; setJustification: (v: string) => void;
   305→  comments: string; setComments: (v: string) => void;
   306→  quotes: TextQuote[]; setQuotes: React.Dispatch<React.SetStateAction<TextQuote[]>>;
   307→  isSubmitted: boolean;
   308→  onSaveDraft: () => void;
   309→  onSubmit: () => void;
   310→  saving: boolean;
   311→  mobileTab: string;
   312→}) {
   313→  const maxTotal = essay.criteria.reduce((sum, c) => sum + c.maxMarks, 0);
   314→  const currentTotal = Object.values(scores).reduce((sum, v) => sum + (v || 0), 0);
   315→  const totalPct = maxTotal > 0 ? (currentTotal / maxTotal) * 100 : 0;
   316→
   317→  const totalColorClass =
   318→    currentTotal > maxTotal
   319→      ? 'text-rose-600'
   320→      : totalPct >= 90
   321→        ? 'text-amber-600'
   322→        : 'text-emerald-600';
   323→
   324→  function updateScore(id: string, value: string) {
   325→    const num = value === '' ? 0 : Math.max(0, Number(value));
   326→    setScores((prev) => ({ ...prev, [id]: num }));
   327→  }
   328→
   329→  function removeQuote(quoteId: string) {
   330→    setQuotes((prev) => prev.filter((q) => q.id !== quoteId));
   331→  }
   332→
   333→  function updateQuoteCriterion(quoteId: string, criterionId: string) {
   334→    setQuotes((prev) => prev.map((q) => q.id === quoteId ? { ...q, criterionId } : q));
   335→  }
   336→
   337→  const allFilled = essay.criteria.every((c) => (scores[c.id] ?? 0) > 0);
   338→  const noOverMax = essay.criteria.every((c) => (scores[c.id] ?? 0) <= c.maxMarks);
   339→  const isValid = allFilled && noOverMax && justification.trim().length > 0;
   340→
   341→  if (mobileTab && mobileTab !== 'evaluation') return null;
   342→
   343→  return (
   344→    <div className="flex flex-col h-full">
   345→      {/* Form Header */}
   346→      <div className="border-b border-slate-200 bg-slate-50 px-4 py-3 shrink-0">
   347→        <div className="flex items-center justify-between mb-2">
   348→          <div className="min-w-0">
   349→            <p className="text-sm font-semibold text-slate-800 truncate">{essay.anonymousId}</p>
   350→            <p className="text-xs text-slate-500 truncate">{essay.competition}</p>
   351→          </div>
   352→          <Badge variant="secondary" className="bg-slate-100 text-slate-600 text-xs shrink-0 ml-2">{essay.category}</Badge>
   353→        </div>
   354→        <div className="flex items-center justify-between">
   355→          <div className="flex items-center gap-2">
   356→            <span className="text-xs text-slate-500">Total Score:</span>
   357→            <span className={`text-lg font-bold ${totalColorClass}`}>{currentTotal}</span>
   358→            <span className="text-xs text-slate-400">/ {maxTotal}</span>
   359→          </div>
   360→          {isSubmitted && (
   361→            <Badge className="bg-emerald-100 text-emerald-700 text-xs gap-1">
   362→              <CheckCircle2 className="h-3 w-3" /> Submitted
   363→            </Badge>
   364→          )}
   365→        </div>
   366→        <Progress value={Math.min(100, totalPct)} className={`h-1.5 mt-1.5 ${currentTotal > maxTotal ? '[&>div]:bg-rose-500' : ''}`} />
   367→      </div>
   368→
   369→      {/* Scrollable Form Content */}
   370→      <ScrollArea className="flex-1">
   371→        <div className="p-4 space-y-5">
   372→          {/* Criteria */}
   373→          <div>
   374→            <div className="flex items-center gap-2 mb-3">
   375→              <ClipboardList className="h-4 w-4 text-emerald-600" />
   376→              <p className="text-sm font-semibold text-slate-700">Evaluation Criteria</p>
   377→            </div>
   378→            <div className="space-y-4">
   379→              {essay.criteria.map((criterion) => {
   380→                const val = scores[criterion.id] ?? 0;
   381→                const overMax = val > criterion.maxMarks;
   382→                const nearMax = val >= criterion.maxMarks * 0.9 && !overMax;
   383→                return (
   384→                  <div key={criterion.id} className="rounded-lg border border-slate-200 p-3">
   385→                    <div className="flex items-start justify-between gap-2 mb-1">
   386→                      <div className="min-w-0">
   387→                        <p className="text-sm font-medium text-slate-700">{criterion.name}</p>
   388→                        {criterion.description && (
   389→                          <p className="text-xs text-slate-400 mt-0.5">{criterion.description}</p>
   390→                        )}
   391→                      </div>
   392→                      <span className="text-xs text-slate-400 shrink-0">Max: {criterion.maxMarks}</span>
   393→                    </div>
   394→                    <div className="flex items-center gap-2 mt-2">
   395→                      <div className="relative flex-1">
   396→                        <Input
   397→                          type="number"
   398→                          min={0}
   399→                          max={criterion.maxMarks}
   400→                          value={val || ''}
   401→                          onChange={(e) => updateScore(criterion.id, e.target.value)}
   402→                          placeholder="0"
   403→                          disabled={isSubmitted}
   404→                          className={`h-8 text-sm pr-12 ${
   405→                            overMax ? 'border-rose-300 focus-visible:ring-rose-400 bg-rose-50/50'
   406→                            : nearMax ? 'border-amber-300 focus-visible:ring-amber-400 bg-amber-50/50'
   407→                            : ''
   408→                          }`}
   409→                        />
   410→                        <span className="absolute right-2.5 top-1/2 -translate-y-1/2 text-xs text-slate-400">/ {criterion.maxMarks}</span>
   411→                      </div>
   412→                    </div>
   413→                    <Textarea
   414→                      placeholder="Brief comment on this criterion..."
   415→                      value={essay.existingScores?.[criterion.id]?.comment ?? ''}
   416→                      disabled={isSubmitted}
   417→                      rows={1}
   418→                      className="mt-2 text-xs resize-none"
   419→                    />
   420→                  </div>
   421→                );
   422→              })}
   423→            </div>
   424→          </div>
   425→
   426→          <Separator />
   427→

   685→// ── Mobile Layout Detector ───────────────────────────────────────────────────
   686→function useIsMobileCustom() {
   687→  const [isMobile, setIsMobile] = useState(false);
   688→  useEffect(() => {
   689→    const mq = window.matchMedia('(max-width: 768px)');
   690→    setIsMobile(mq.matches);
   691→    const handler = (e: MediaQueryListEvent) => setIsMobile(e.matches);
   692→    mq.addEventListener('change', handler);
   693→    return () => mq.removeEventListener('change', handler);
   694→  }, []);
   695→  return isMobile;
   696→}

   726→  const [selectedId, setSelectedId] = useState<string | null>(null);

   729→

   732→  const [currentPage, setCurrentPage] = useState(1);
   733→

   735→  const [scores, setScores] = useState<Record<string, number>>({});

   737→  const [comments, setComments] = useState('');

   739→  const [saving, setSaving] = useState(false);
   740→  const [submitDialogOpen, setSubmitDialogOpen] = useState(false);
   741→  const [isSubmitted, setIsSubmitted] = useState(false);
   742→
   743→  // Mobile state
   744→  const isMobile = useIsMobileCustom();
   745→  const [mobileTab, setMobileTab] = useState('list');
   746→
   747→  // Load essays
   748→  useEffect(() => {
   749→    async function load() {
   750→      try {
   751→        const res = await fetch('/api/seed?action=examiner-workspace');
   752→        if (res.ok) {
   753→          const json = await res.json();
   754→          if (json.data) { setEssays(json.data); setLoading(false); return; }
   755→        }
   756→      } catch { /* fall through */ }
   757→      setEssays(MOCK_ESSAYS);

   765→    if (!selectedId) return null;
   766→    const essay = essays.find((e) => e.id === selectedId);

   768→    return {
   769→      ...essay,
   770→      criteria: MOCK_CRITERIA,
   771→      existingScores: essay.status === 'IN_PROGRESS' ? MOCK_EXISTING_SCORES : undefined,

   831→      setEssays((prev) =>
   832→        prev.map((e) => (e.id === selectedId ? { ...e, status: 'SUBMITTED' as EvaluationStatus } : e))
   833→      );
   834→    }
