    95→    95→function generatePageLines(page: number) {
    96→    96→  const title = page === 1;
    97→    97→  const lines: { w: number; h: number; mt: boolean }[] = [];
    98→    98→  const seed = page * 7;
    99→    99→  if (title) {
   100→   100→    lines.push({ w: 65, h: 12, mt: true });
   101→   101→    lines.push({ w: 40, h: 8, mt: true });
   102→   102→    lines.push({ w: 0, h: 16, mt: false });
   103→   103→  }
   104→   104→  const lineCount = title ? 18 : 22;
   105→   105→  for (let i = 0; i < lineCount; i++) {
   106→   106→    const variance = ((seed + i * 13) % 30) - 10;
   107→   107→    lines.push({ w: Math.max(50, Math.min(95, 80 + variance)), h: 5, mt: i === 0 && !title });
   108→   108→  }
   109→   109→  return lines;
   110→   110→}

   296→   296→function EvaluationForm({
   297→   297→  essay, scores, setScores, justification, setJustification,
   298→   298→  comments, setComments, quotes, setQuotes, isSubmitted,
   299→   299→  onSaveDraft, onSubmit, saving, mobileTab
   300→   300→}: {
   301→   301→  essay: EssayDetail;
   302→   302→  scores: Record<string, number>;
   303→   303→  setScores: React.Dispatch<React.SetStateAction<Record<string, number>>>;
   304→   304→  justification: string; setJustification: (v: string) => void;
   305→   305→  comments: string; setComments: (v: string) => void;
   306→   306→  quotes: TextQuote[]; setQuotes: React.Dispatch<React.SetStateAction<TextQuote[]>>;
   307→   307→  isSubmitted: boolean;
   308→   308→  onSaveDraft: () => void;
   309→   309→  onSubmit: () => void;
   310→   310→  saving: boolean;
   311→   311→  mobileTab: string;
   312→   312→}) {
   313→   313→  const maxTotal = essay.criteria.reduce((sum, c) => sum + c.maxMarks, 0);
   314→   314→  const currentTotal = Object.values(scores).reduce((sum, v) => sum + (v || 0), 0);
   315→   315→  const totalPct = maxTotal > 0 ? (currentTotal / maxTotal) * 100 : 0;
   316→   316→
   317→   317→  const totalColorClass =
   318→   318→    currentTotal > maxTotal
   319→   319→      ? 'text-rose-600'
   320→   320→      : totalPct >= 90
   321→   321→        ? 'text-amber-600'
   322→   322→        : 'text-emerald-600';
   323→   323→
   324→   324→  function updateScore(id: string, value: string) {
   325→   325→    const num = value === '' ? 0 : Math.max(0, Number(value));
   326→   326→    setScores((prev) => ({ ...prev, [id]: num }));
   327→   327→  }
   328→   328→
   329→   329→  function removeQuote(quoteId: string) {
   330→   330→    setQuotes((prev) => prev.filter((q) => q.id !== quoteId));
   331→   331→  }
   332→   332→
   333→   333→  function updateQuoteCriterion(quoteId: string, criterionId: string) {
   334→   334→    setQuotes((prev) => prev.map((q) => q.id === quoteId ? { ...q, criterionId } : q));
   335→   335→  }
   336→   336→
   337→   337→  const allFilled = essay.criteria.every((c) => (scores[c.id] ?? 0) > 0);
   338→   338→  const noOverMax = essay.criteria.every((c) => (scores[c.id] ?? 0) <= c.maxMarks);
   339→   339→  const isValid = allFilled && noOverMax && justification.trim().length > 0;
   340→   340→
   341→   341→  if (mobileTab && mobileTab !== 'evaluation') return null;
   342→   342→
   343→   343→  return (
   344→   344→    <div className="flex flex-col h-full">
   345→   345→      {/* Form Header */}
   346→   346→      <div className="border-b border-slate-200 bg-slate-50 px-4 py-3 shrink-0">
   347→   347→        <div className="flex items-center justify-between mb-2">
   348→   348→          <div className="min-w-0">
   349→   349→            <p className="text-sm font-semibold text-slate-800 truncate">{essay.anonymousId}</p>
   350→   350→            <p className="text-xs text-slate-500 truncate">{essay.competition}</p>
   351→   351→          </div>
   352→   352→          <Badge variant="secondary" className="bg-slate-100 text-slate-600 text-xs shrink-0 ml-2">{essay.category}</Badge>
   353→   353→        </div>
   354→   354→        <div className="flex items-center justify-between">
   355→   355→          <div className="flex items-center gap-2">
   356→   356→            <span className="text-xs text-slate-500">Total Score:</span>
   357→   357→            <span className={`text-lg font-bold ${totalColorClass}`}>{currentTotal}</span>
   358→   358→            <span className="text-xs text-slate-400">/ {maxTotal}</span>
   359→   359→          </div>
